# Jaipur's six goods and its bonus mark, generated into
# src/apps_local/jaipur/JaipurGoods.h by tools_local/gen_jaipur_art.sh.
# Format: alias = lucide-name.
#
# Separate from tools_local/icons.txt because these are the only icons in the
# fork that need a 56px cut, for the market cards, and asking the shared
# generator for it produced a 56px copy of all 42 shelf icons.
#
# Measured, not assumed: that cost **no flash at all**. The arrays are
# `static const` in a header, so every translation unit that does not name one
# drops it, and the firmware came out byte-identical either way. What it did
# cost was 58KB of generated source in a header every app includes, and the
# provenance boundary -- these are Lucide's, the camel next door is ours.
#
good_diamond = diamond      # a clean rhombus; gem's facet lines are interior
                            # detail and mush at 32px, which is what ruled out
                            # gamepad-2 and castle for the shelf
good_gold    = coins        # a stack. Value reads as quantity
good_silver  = layers       # stacked bars, which is how silver is actually
                            # traded. Three flat shapes cannot be confused with
                            # the diamond's single upright rhombus, and unlike a
                            # coin it does not collide with the gold stack.
                            # `medal` was the first pick and was rejected on
                            # sight: a ribbon over a disc over a star is three
                            # shapes inside 32px and it reads as a prize.
                            # Runner-up was `weight`; also rejected: `badge` and
                            # `disc` (too close to the gold coins), `hexagon`
                            # (abstract, says nothing about metal), `stamp`
                            # (reads as office stationery)
good_cloth   = shirt
good_spice   = leaf
good_leather = backpack     # Lucide has no hide or boot; a bag is the leather
                            # good a market would actually sell

# The bonus token, which is not a good and should not look like one.
bonus_token  = sparkles
